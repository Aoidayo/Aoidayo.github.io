二分查找

在有序数组上快速查找一个元素的位置，或者查找指定元素的位置。

  • le，left exclude，小于x的最后一个
  • li，left include，小于等于x的最后一个
  • re，right exclude，大于x的第一个
  • ri，right include，大于等于x的第一个

package template;

/**
 * cate: ["二分查找"]
 */
public class bisearch {

    /**
     * le: left exclude, <x的最后一个
     * @param q
     * @return
     */
    public int le(int[] q, int x){
        int l=0, r=q.length-1,m;
        while(l<=r){ // 维护[l,r]区间
            m = l+(r-l)/2;
            if(q[m] >= x) r=m-1; // r缩到<x的最后一个
            else l=m+1;
        }
        return r;
    }


    /**
     * li left include: <=x的最后一个
     * @param q
     * @param x
     * @return
     */
    public int li(int[]q, int x){
        int l=0, r=q.length-1,m;
        while(l<=r){ // [l,r]
            m = l+(r-l)/2;
            if(q[m]>x) r = m-1; // r缩到<=x的第一个
            else l=m+1;
        }
        return r;
    }

    public int re(int[]q, int x){
        // right exclude >x的第一个
        int l=0,r=q.length-1,m;
        while(l<=r){
            m = l+ (r-l)/2;
            if(q[m]<=x) l = m+1; // l缩到>x的第一个，跳过所有<=x的
            else r = m-1;
        }
        return l;
    }

    /**
     * right include
     * @param q
     * @param x
     * @return
     */
    public int ri(int[]q, int x){
        // right include >=x的第一个
        int l=0,r=q.length-1,m;
        while(l<=r){
            m = l+ (r-l)/2;
            if(q[m]<x) l = m+1; // l缩到>=x的第一个，跳过所有<x的
            else r = m-1;
        }
        return l;
    }





}

/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public ListNode reverseList(ListNode head) {
        if(head == null || head.next == null) return head;
        // 不能写 ListNode pre = head, cur = head.next;
        ListNode pre = null, cur = head;
        while(cur != null){
            // 
            ListNode nex = cur.next;
            cur.next = pre;
            pre = cur;
            cur = nex;
        }
        // cur == null, pre是最后一个
        return pre;
    }
}

指定l，r位置的链表反转

/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public ListNode reverseBetween(ListNode head, int left, int right) {
        if(left == right) return head;
        ListNode dummy = new ListNode(); // 哨兵
        dummy.next = head;
        ListNode p0 = dummy;

        // 找l-1, 然后在[l,r]上做r-l次反转
        for(int i=0;i<left-1;i++){ // 第一个的话，l=1, p0做0次到dummy
            p0 = p0.next;
        }
        ListNode pre = p0.next, cur = p0.next.next, nxt=null;
        for(int i=0;i<right-left;i++){ // r-l次反转
            nxt = cur.next;
            cur.next = pre;
            pre = cur;
            cur = nxt;
        }
        // pre是r的位置，cur是原r的后一个
        // p0.next = l, l的下一个结上cur
        p0.next.next = cur;
        p0.next = pre;

        return dummy.next;

    }
}

public class Solution {
    public boolean hasCycle(ListNode head) {
        if(head == null || head.next == null) return false;
        ListNode slow = head, fast = head;
        while(fast.next.next != null){
            fast = fast.next.next;
            slow = slow.next;
            if(slow == fast) return true;
        }
        return false;
    }
}

环形链表入环位置：

public class Solution {
    public ListNode detectCycle(ListNode head) {
        if(head == null || head.next == null) return null;
        ListNode slow = head, fast = head;
        if(fast.next == null || fast.next.next == null) return null;
        // 相遇点
        while(fast!=null && fast.next!=null){
            fast = fast.next.next;
            slow = slow.next;
            if(fast == slow) break;
        }
        if(fast == null || fast.next == null) return null;
        // fast == slow
        ListNode p0 = head;
        while(p0 != slow){
            p0 = p0.next;
            slow = slow.next;
        }
        return slow;
    } 

}

Q：冒泡排序？
A：

每躺交换一个到最后面，依次递减，n个元素交换n-1次到最后。

public void sort(int[] q){
	int n = q.length;
	for(int i=0;i<n-1;i++){
		// n-1-i 与后一个交换
		for(int j=0;j<n-1-i; j++){
			if(q[j] > q[j+1]){
				int tmp = q[j]; q[j] = q[j+1]; q[j+1] = tmp;
			}
		}
	}
	// done
}
Q：基准排序？快速排序？
A：

先取pivot基准点，围绕pivot使两边相对pivot有序，接着递归处理。

注意使用do while写会简单点。


public int[] sort_quikc_outer(int[] q){
	sort_quikc(q, 0, q.length-1);
	return q;
}

public void sort_quick(int[] q, int l, int r){

	if(l>=r) return; 
	int i = l-1, j = r+1, x = q[l + (r-l)/2];
	
	// 使[l,r]围绕pivot 相对有序
	while(i<j){
		do i++; while(q[i]<x); // >=x的交换到右边，最后i及其右侧>=x
		do j--; while(q[j]>x); // <=x的交换到左边，最后j及其左侧<=x
		if(i<j){
			int tmp = q[i]; q[i] = q[j]; q[j] = tmp;
		}
	}
	
	// 分治处理 [l, j]<=x, [j+1,r]>=x
	sort_quick(q, l, j); sort_quick(q, j+1, r);
}
Q：快速排序如果不用do while怎么写？
A：

如果不写do while的code如下：


public int[] sort_quikc_outer(int[] q){
	sort_quikc(q, 0, q.length-1);
	return q;
}

public void sort_quick(int[] q, int l, int r){

	if(l>=r) return; 
	int i = l-1, j = r+1, x = q[l + (r-l)/2];
	
	// 使[l,r]围绕pivot 相对有序
	while(i<j){
		while(q[i]<x) i++; // >=x的交换到右边
		while(q[j]>x) j--; // <=x的交换到左边
		if(i<j){
			int tmp = q[i]; q[i] = q[j]; q[j] = tmp;
			// 注意，这里需要再做一次 i++, j--;
			i++; j--; 
		}
	}
	
	// 分治处理 [l, j]<=x, [j+1,r]>=x
	sort_quick(q, l, j); sort_quick(q, j+1, r);
}

注意讨论一个特殊情况，q[i] == q[j] == x的时候，上面还是会做i++, j–不会循环卡住。

简单讨论一下为什么不建议写do while，如果直接这么写：

while(i<j){
	while(q[i] < x) i++; // >=x的交换到右边
	while(q[j] > x) j--; // <=x的交换到左边
	if(i<j){
		swap(q[i], q[j]);
	}
}

会出现在i<j，q[i] = q[j] = x的时候，交换了q[i], q[j]，但是下一次while q[i] <x, q[j] >x的时候ij没有变化，交换的时候也没有变化，于是就导致死循环了。

所以需要在swap之后做i++, j--。

如果是do while

while(i<j){
	do i++; while(q[i] < x);  // >=x的交换到右边
	do j--; while(q[j] > x); // <=x的交换到左边
	if(i<j){
		swap(q[i], q[j]);
	}
}

// 在第4行 i<j的时候，q[i] == q[j] = x, 交换之后
// do i++ 和 do j--都是继续先执行再判断的，所以不会死循环。
Q：归并排序？
A：

先分治递归，在合并处理两个有序数组为一个有序数组，最后返回。

public int[] sort_merge_outer(int[] q){
	sort_merge(q, 0, q.length-1);
	return q;
}

public void sort_merge(int[] q, int l, int r){
	if(l >= r) return;
	// 分治
	int m = l + (r-l)/2;
	sort_merge(q, l, m); sort_merge(q, m+1, r);
	
	// 合并有序数组
	int[] a = new int[r-l+1];
	int i = l, j = m+1;
	int k = 0;
	while(i<=m && j<=r){
		if(q[i]<q[j]) a[k++] = q[i++];
		else a[k++] = q[j++];
	}
	while(i<=m) a[k++] = q[i++];
	while(j<=r) a[k++] = q[j++];
	// 赋值回原数组的[l,r]
	k = 0; i = l;
	for(; i<=r;) q[i++] = a[k++];
}
Q：快速排序和归并排序哪个更加稳定？
A：

归并无论是相同大小的前后元素保持，还是最差时间复杂度，都更加稳定。

归并更稳定，如果是[1, 2a, 2b, 3, 4]，那么排序后2a肯定在2b前面，快速不一定；同时归并时时间复杂度最坏也是o(nlgn)。

快速排序 如果每次选的pivot是最小的话，那么分层就是n层，每次需要比较的元素数量是n, n-1, n-2, ...,1，其和是$n^2$。

正常选择中间值，递归lgn层，每层比较n个元素，最后其和就是nlgn。

前缀和

$$ S[0] = 0 \ S[i] = S[i-1] + a[i-1] \ $$

其中 $S[i]$ 表示前 $i-1$个元素的和，所以 $len(S) = n+1$，最后一个元素$S[n] = 0 + a[0] + a[1] + … + a[n-1]$ 为前$n-1$个元素之和。

$l$至$r$区间的元素和：$sum(a[l,…, r]) = S[r+1] - S[l]$，所有位置统一的区间和，不需要特判 $l=0$ 的情况。

$S[0]=0$ 作为占位的哨兵使用，没有任何含义，方便我们处理数组为空、包含头的区域和。

例如 数组为空时，$a[] = [], S[0]=0$。

例如，包含头的区域和：$a[] = [1,2,3,4,5], S[] = [0, 1, 3, 6, 10, 15]$，在求$[0, 3]$区域和可用$S[4]-S[0]$表示。如果没有$S[0]$，那么$S[] = [1,3,6,10,15]$，$S[i]$表示前i个元素之和，求$[0,3]$需要表示为$S[3] - 0$。

原始的前缀和是：

$$ S[i] = S[i-1] + A[i] \ S[0] = A[0] $$

$l$~~~~至~~~~$r$~~~~区间的元素和：

$$ sum(a[l,…, r]) = S[r] - S[l-1], when \ l>=1; \ sum(a[0,…,r]) = S[r] - 0, when \ l=0; $$

差分

通过前后元素之差，快速进行区域加和。

给一个例子：$a[] = [2,3,4,5,6]$，其差分是$diff[] = [2,1,1,1,1]$，$diff$数组和输入数组$a$长度相同，同时$diff[0] = a[0]$、$diff[i] = a[i]-a[i-1]$。

$diff[i]$ 表示当前$i$元素和前一个$i-1$元素之差。

$diff[0]=a[0]$ 用于将差分数组恢复为原始数组。

接着，思考如下两个问题：

  1. 如何从diff恢复a。
  2. 如何快速加减指定区间。
Q：如何构造diff数组？
A：
public void diff(int[] q){
	int n = q.length;
	int[] diff = new int[n];
	diff[0] = q[0];
	for(int i=1;i<n;i++) diff[i] = q[i] - q[i-1];
}
Q：如何从diff恢复a？
A：
public void diffToA(int[] diff){
	int n = diff.length;
	int[] a = new int[n];
	a[0] = diff[0];
	// a[i] = a[i-1] + diff[i] = ( a[i-1] + a[i] - a[i-1] ) = a[i]
	for(int i=1;i<n;i++) a[i] = a[i-1] + diff[i];
}
Q：如何对指定$[l,r]$区间进行快速加减？
A：

$[l,r]$区间$+x$：$diff[l]+x, diff[r+1]-x$，在恢复原数组时从$[l,r]$区间上每个元素都加了x，$r+1$这个位置上恢复原来的大小。

减法同理。

diff[l] += x;
diff[r+1] -= x;

讨论下边界问题，$l\in[0,n-1],\ r\in[0,n-1]$，但是$r+1 \in[1, n]$，如果区间快速加减涉及到r的问题，那么如果r是n-1的时候，只需要做$diff[l] +x$。

总结下：

diff[l] += x;
// （1）当r<n-1时
diff[r+1] -= x; 
// （2）当r=n-1时，不操作。

并查集

通过手动的方式构建一个支持查询、合并的集合。

合并时记录各个分块集合（连通块）的大小，查询时可以改善原来的查询路径。

思考两种方式实现的并查集有何区别？

  • 基于数组实现的并查集。
  • 基于hashmap实现的并查集。

例题：

  • 128. 最长连续序列 - 力扣（LeetCode）
Q：基于数组的并查集？
A：

基于数组的并查集，预设当前集合中存在$0 \sim n-1$这$n$个元素，我们对集合的并查操作范围在这个n个数范围内。

package template;

public class union_find {

    int[] fa; //  代表元 数组，初始每个元素都是对应集合的代表元
    int[] size; // 初始都为1
    int cc; // 连通块个数，初始为n

    public union_find(int n){
        fa = new int[n];
        size = new int[n];
        for(int i=0;i<n;i++) {
            fa[i] = i;
            size[i] = 1;
        }
        cc = n; // 初始化n个连通块
    }

    /**
     * 查找x所在集合的 代表元
     * 代表元的性质：fa[x] == x
     *
     * Note: 如果find(int x)出现递归错误/stackoverflow，那么很有可能是修改非直接元的逻辑写错了，写成修改代表元的了。
     * @param x
     * @return
     */
    public int find(int x){
        // 如果fa[x] == x, 那么直接返回fa[x] 即可
        // 如果fa[x]!=x, 那么需要向fa上面追溯，同时顺道将路上的非直接元的代表元修改为深度1的直接元
        if(fa[x] != x) fa[x] = find(fa[x]);
        return fa[x];
    }

    /**
     * 将from集合合并到to集合上
     * @param from
     * @param to
     * @return
     *  fasle 合并失败（包括from和to就是同一个集合的情况）
     *  true 合并成功
     */
    public boolean merge(int from, int to){
        int x = find(from), y = find(to);
        if(x == y) return false; // 来自同一个元
        fa[x] = y; // 合并
        size[y] += size[x]; // size
        cc--;
        return true;
    }

    /**
     * x 和 y是否是一个集合
     * @param x
     * @param y
     * @return
     */
    public boolean is_same(int x, int y){
        return find(x) == find(y);
    }


    /**
     * 返回x所在集合的size
     */
    public int get_size(int x){
        // size[ x所在集合的代表元 ]
        return size[find(x)];
    }



}
Q：基于hashmap的并查集？为什么需要有基于hashmap的并查集？
A：

基于数组的并查集，其集合范围在0~n-1这n个元素上。使用hashmap构建fa和sz，可以不限制集合元素的范围，更加灵活。

static class union_find_map{
	public HashMap<Integer, Integer> fa = new HashMap<>(); // 代表元
	public HashMap<Integer, Integer> sz = new HashMap<>(); // 初始为1
	int cc;

	public int find(int x){
		// 如果x不存在与Hashmap，先创建；
		createIfAbsent(x);
		// 存在
		if(fa.get(x) != x) fa.put(x, find(fa.get(x)));
		return fa.get(x);
	}

	public void createIfAbsent(int x){
		if(!fa.containsKey(x)){
			fa.put(x, x);
			sz.put(x, 1);
			cc++;
		}
	}

	public boolean merge(int from , int to){
		// 如果 from 和 to 不存在，先创建
		createIfAbsent(from);
		createIfAbsent(to);
		int x = find(from), y = find(to);
		if(x == y) return false; //  不需要合并
		fa.put(x, y);
		sz.put(y, sz.get(y)+ sz.get(x));
		cc--; //
		return true; //
	}

	public int sz(int x){
		return sz.get(find(x));
	}

}
